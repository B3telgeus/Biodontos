-- ----------------------------------------------------------------------------
-- 1. TABLA: DOCTORS (Profesionales Odontológicos)
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.doctors (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid UNIQUE,
  first_name text NOT NULL,
  last_name text NOT NULL,
  email text NOT NULL UNIQUE,
  phone text,
  license_number text NOT NULL UNIQUE,
  specialty text NOT NULL DEFAULT 'Odontología General',
  signature_stamp_url text,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT doctors_pkey PRIMARY KEY (id),
  CONSTRAINT doctors_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE CASCADE
);

-- ----------------------------------------------------------------------------
-- 2. TABLA: PATIENTS (Datos de Identificación del Paciente)
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.patients (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  user_id uuid UNIQUE,
  first_name character varying(50) NOT NULL,
  first_last_name character varying(50) NOT NULL DEFAULT '',
  second_last_name character varying(50) NOT NULL DEFAULT '',
  document_type_id integer NOT NULL DEFAULT 1,
  dni_cedula character varying(20) NOT NULL UNIQUE,
  birth_date date NOT NULL DEFAULT CURRENT_DATE,
  birth_place character varying(50) NOT NULL DEFAULT 'No especificado',
  eps character varying(50) NOT NULL DEFAULT 'Particular',
  occupation character varying(50),
  marital_status character varying(30),
  blood_group character varying(2) NOT NULL DEFAULT 'O',
  rh_factor character varying(2) NOT NULL DEFAULT '+',
  phone character varying(20) NOT NULL DEFAULT '',
  email text,
  address text,
  guardian_relationship character varying(30),
  guardian_first_name character varying(50),
  guardian_last_name character varying(50),
  guardian_phone character varying(20),
  guardian_address character varying(150),
  last_dental_visit character varying(150),
  consultation_reason character varying(250),
  current_consultation_reason character varying(250),
  illness_history text,
  critical_allergies text,
  allergies_description text,
  consent_signed boolean DEFAULT false,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT patients_pkey PRIMARY KEY (id),
  CONSTRAINT patients_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE SET NULL
);

-- ----------------------------------------------------------------------------

-- ----------------------------------------------------------------------------

  id uuid NOT NULL DEFAULT gen_random_uuid(),
  patient_id uuid NOT NULL UNIQUE,
  tratamiento_medico_actual boolean NOT NULL DEFAULT false,
  toma_medicamentos boolean NOT NULL DEFAULT false,
  alergias boolean NOT NULL DEFAULT false,
  cardiopatias boolean NOT NULL DEFAULT false,
  alteracion_presion_arterial boolean NOT NULL DEFAULT false,
  embarazo boolean NOT NULL DEFAULT false,
  diabetes boolean NOT NULL DEFAULT false,
  hepatitis boolean NOT NULL DEFAULT false,
  irradiaciones boolean NOT NULL DEFAULT false,
  discrasias_sanguineas boolean NOT NULL DEFAULT false,
  fiebre_reumatica boolean NOT NULL DEFAULT false,
  enfermedades_renales boolean NOT NULL DEFAULT false,
  inmunosupresion boolean NOT NULL DEFAULT false,
  trastornos_emocionales boolean NOT NULL DEFAULT false,
  patologia_respiratoria boolean NOT NULL DEFAULT false,
  trastornos_gastricos boolean NOT NULL DEFAULT false,
  epilepsia boolean NOT NULL DEFAULT false,
  cirugias boolean NOT NULL DEFAULT false,
  enfermedades_orales boolean NOT NULL DEFAULT false,
  otras_alteraciones boolean NOT NULL DEFAULT false,
  observaciones character varying(200) NOT NULL DEFAULT '',
  antecedentes_familiares character varying(200) NOT NULL DEFAULT '',

  created_at timestamp with time zone DEFAULT now(),

-- ----------------------------------------------------------------------------
-- 4. TABLA: CLINICAL_RECORDS (Historia Clínica, Exámenes, DP/DR y Cierre Legal)
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.clinical_records (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  patient_id uuid NOT NULL UNIQUE,
  dentition_type dentition_type DEFAULT 'ADULT',
  main_diagnosis text,
  related_diagnoses text,

  status clinical_record_status NOT NULL DEFAULT 'ABIERTA',

  closed_by_doctor_id uuid,

  closure_signature_url text,

  CONSTRAINT clinical_records_pkey PRIMARY KEY (id),
  CONSTRAINT clinical_records_patient_id_fkey FOREIGN KEY (patient_id) REFERENCES public.patients(id) ON DELETE CASCADE,
  CONSTRAINT clinical_records_closed_by_doctor_id_fkey FOREIGN KEY (closed_by_doctor_id) REFERENCES public.doctors(id)

-- ----------------------------------------------------------------------------
-- 5. TABLA: INFORMED_CONSENTS (Consentimiento Legal de Operatoria Dental)

-- ----------------------------------------------------------------------------

-- ----------------------------------------------------------------------------
-- 8. TABLA: EVOLUTIONS (Evolución de Tratamiento, Valores, Abonos y Firmas)
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.evolutions (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  record_id uuid NOT NULL,
  doctor_id uuid,

  procedure_time time without time zone DEFAULT CURRENT_TIME,

-- ----------------------------------------------------------------------------
-- 13. TABLA: PERIODONTAL_SITES (Sondaje de 6 Sitios con Cálculo Automático de CAL)
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.periodontal_sites (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  examination_id uuid NOT NULL,
  tooth_number integer NOT NULL,
  site_location periodontal_site_enum NOT NULL,
  pocket_depth_mm numeric(3, 1) NOT NULL DEFAULT 1.0,
  gingival_recession_mm numeric(3, 1) NOT NULL DEFAULT 0.0,
  clinical_attachment_loss_mm numeric(4, 1) GENERATED ALWAYS AS (pocket_depth_mm + gingival_recession_mm) STORED,
  bleeding_on_probing boolean NOT NULL DEFAULT false,
  suppuration boolean NOT NULL DEFAULT false,
  mobility_grade integer NOT NULL DEFAULT 0,
  furcation_grade integer NOT NULL DEFAULT 0,
  CONSTRAINT periodontal_sites_pkey PRIMARY KEY (id),
  CONSTRAINT periodontal_sites_examination_id_fkey FOREIGN KEY (examination_id) REFERENCES public.periodontal_examinations(id) ON DELETE CASCADE,
  CONSTRAINT periodontal_sites_tooth_site_unique UNIQUE (examination_id, tooth_number, site_location),
  CONSTRAINT check_mobility_grade CHECK (mobility_grade BETWEEN 0 AND 3),
  CONSTRAINT check_furcation_grade CHECK (furcation_grade BETWEEN 0 AND 4)
);

-- 18. AUTOMATIZACIÓN DE AUTENTICACIÓN (Trigger handle_new_user)
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger AS $$
DECLARE
  user_role text;
BEGIN
  user_role := COALESCE(new.raw_user_meta_data->>'role', 'PATIENT');

  IF UPPER(user_role) = 'DOCTOR' THEN
    INSERT INTO public.doctors (
      user_id,
      first_name,
      last_name,
      email,
      phone,
      license_number,
      specialty
    ) VALUES (
      new.id,
      COALESCE(new.raw_user_meta_data->>'first_name', ''),
      COALESCE(new.raw_user_meta_data->>'last_name', ''),
      new.email,
      new.raw_user_meta_data->>'phone',
      COALESCE(new.raw_user_meta_data->>'license_number', 'PENDIENTE-' || SUBSTRING(new.id::text, 1, 8)),
      COALESCE(new.raw_user_meta_data->>'specialty', 'Odontología General')
    )
    ON CONFLICT (license_number) DO NOTHING;

  ELSE
    -- Registro de Paciente
    INSERT INTO public.patients (
      user_id,
      first_name,
      first_last_name,
      second_last_name,
      dni_cedula,
      email,
      phone,
      birth_date,
      birth_place,
      eps,
      blood_group,
      rh_factor,
      consent_signed
    ) VALUES (
      new.id,
      COALESCE(new.raw_user_meta_data->>'first_name', ''),
      COALESCE(new.raw_user_meta_data->>'last_name', ''),
      COALESCE(new.raw_user_meta_data->>'second_last_name', ''),
      COALESCE(new.raw_user_meta_data->>'dni_cedula', 'TEMP-' || SUBSTRING(new.id::text, 1, 8)),
      new.email,
      COALESCE(new.raw_user_meta_data->>'phone', ''),
      COALESCE((new.raw_user_meta_data->>'birth_date')::date, CURRENT_DATE),
      COALESCE(new.raw_user_meta_data->>'birth_place', 'No especificado'),
      COALESCE(new.raw_user_meta_data->>'eps', 'Particular'),
      COALESCE(new.raw_user_meta_data->>'blood_group', 'O'),
      COALESCE(new.raw_user_meta_data->>'rh_factor', '+'),
      true
    )
    ON CONFLICT (dni_cedula) DO NOTHING;
  END IF;

  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

CREATE OR REPLACE FUNCTION public.is_doctor()
RETURNS boolean AS $$
BEGIN
  RETURN EXISTS (
    SELECT 1 FROM public.doctors WHERE user_id = auth.uid()
  );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Políticas para PATIENTS
DROP POLICY IF EXISTS "Doctores tienen acceso completo a pacientes" ON public.patients;
CREATE POLICY "Doctores tienen acceso completo a pacientes"
  ON public.patients FOR ALL
  TO authenticated
  USING (public.is_doctor());

DROP POLICY IF EXISTS "Pacientes pueden ver su propio perfil" ON public.patients;
CREATE POLICY "Pacientes pueden ver su propio perfil"
  ON public.patients FOR SELECT
  TO authenticated
  USING (user_id = auth.uid());

DROP POLICY IF EXISTS "Pacientes pueden actualizar sus datos personales" ON public.patients;
CREATE POLICY "Pacientes pueden actualizar sus datos personales"
  ON public.patients FOR UPDATE
  TO authenticated
  USING (user_id = auth.uid());

-- Políticas para HISTORIAS Y EVOLUCIONES
DROP POLICY IF EXISTS "Doctores gestionan historias clínicas" ON public.clinical_records;
CREATE POLICY "Doctores gestionan historias clínicas"
  ON public.clinical_records FOR ALL
  TO authenticated
  USING (public.is_doctor());

DROP POLICY IF EXISTS "Pacientes pueden consultar su propia historia clínica" ON public.clinical_records;
CREATE POLICY "Pacientes pueden consultar su propia historia clínica"
  ON public.clinical_records FOR SELECT
  TO authenticated
  USING (patient_id IN (SELECT id FROM public.patients WHERE user_id = auth.uid()));

DROP POLICY IF EXISTS "Doctores gestionan evoluciones" ON public.evolutions;
CREATE POLICY "Doctores gestionan evoluciones"
  ON public.evolutions FOR ALL
  TO authenticated
  USING (public.is_doctor());

DROP POLICY IF EXISTS "Pacientes pueden ver sus evoluciones" ON public.evolutions;
CREATE POLICY "Pacientes pueden ver sus evoluciones"
  ON public.evolutions FOR SELECT
  TO authenticated
  USING (record_id IN (
    SELECT id FROM public.clinical_records 
    WHERE patient_id IN (SELECT id FROM public.patients WHERE user_id = auth.uid())
  ));

DROP POLICY IF EXISTS "Doctores gestionan sitios periodontales" ON public.periodontal_sites;
CREATE POLICY "Doctores gestionan sitios periodontales"
  ON public.periodontal_sites FOR ALL
  TO authenticated
  USING (public.is_doctor());

DROP POLICY IF EXISTS "Pacientes pueden consultar sus sitios periodontales" ON public.periodontal_sites;
CREATE POLICY "Pacientes pueden consultar sus sitios periodontales"
  ON public.periodontal_sites FOR SELECT
  TO authenticated
  USING (examination_id IN (
    SELECT pe.id FROM public.periodontal_examinations pe
    JOIN public.clinical_records cr ON pe.record_id = cr.id
    WHERE cr.patient_id IN (SELECT id FROM public.patients WHERE user_id = auth.uid())
  ));
